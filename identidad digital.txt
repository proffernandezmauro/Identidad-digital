        function switchTab(tab) {
            const btnIntro = document.getElementById('tab-btn-intro');

            const inactiveClass = "flex-1 py-3 text-center border-b-2 border-transparent text-gray-500 hover:text-gray-800 dark:hover:text-gray-200 flex items-center justify-center gap-2 transition";

            if (tab === 'intro') {

                contentRef.classList.add('hidden');
                contentCust.classList.add('hidden');

                btnRef.className = inactiveClass;

            } else if (tab === 'reflection') {
                contentIntro.classList.add('hidden');
                contentRef.classList.remove('hidden');
                contentCust.classList.add('hidden');

                btnIntro.className = inactiveClass;
                btnRef.className = activeClass;
                btnCust.className = inactiveClass;

                contentIntro.classList.add('hidden');
                contentRef.classList.add('hidden');
                contentCust.classList.remove('hidden');

                btnIntro.className = inactiveClass;
                btnRef.className = inactiveClass;
                btnCust.className = activeClass;
            }

        function updateAllBadgesAndProgress() {
            const keys = Object.keys(consignasData);
            let answeredCount = 0;

            keys.forEach(key => {
                const hasAnswer = userResponses[key] && userResponses[key].trim().length > 0;
                if (hasAnswer) answeredCount++;

                if (key === 'avatar') {
                    const badge = document.getElementById('badge-avatar');
                    if (badge) badge.className = `absolute bottom-0 right-0 ${hasAnswer ? 'bg-emerald-500' : 'bg-brand-600'} text-white p-2 rounded-full shadow-lg border-2 border-white dark:border-gray-900 flex items-center justify-center`;
                }
                if (key === 'followers') {
                    const icon = document.getElementById('badge-followers-icon');
                    if (icon) icon.className = `ph-bold ${hasAnswer ? 'ph-check-circle text-emerald-500' : 'ph-circle text-gray-400'} text-xs`;
                }
                if (key === 'following') {
                    const icon = document.getElementById('badge-following-icon');
                    if (icon) icon.className = `ph-bold ${hasAnswer ? 'ph-check-circle text-emerald-500' : 'ph-circle text-gray-400'} text-xs`;
                }
                if (key === 'bio') {
                    const badge = document.getElementById('badge-bio');
                    if (badge) {
                        badge.innerText = hasAnswer ? '✓ Respondido' : 'Reflexionar Bio';
                        badge.className = `text-xs px-2 py-1 ${hasAnswer ? 'bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-300' : 'bg-gray-200 dark:bg-gray-700 text-gray-700 dark:text-gray-300'} rounded font-medium`;
                    }
                }
                if (key.startsWith('story')) {
                    const icon = document.getElementById(`badge-${key}-icon`);
                    if (icon) icon.className = `ph-bold ${hasAnswer ? 'ph-check-circle text-emerald-500' : 'ph-circle text-gray-400'} text-[10px]`;
                }
            });

            const total = keys.length;
            const percentage = Math.round((answeredCount / total) * 100);

            document.getElementById('badge-progress-tab').innerText = `${answeredCount}/${total}`;
            document.getElementById('progress-percentage').innerText = `${percentage}%`;
            document.getElementById('summary-completion-status').innerText = `${answeredCount} de ${total} Preguntas Completadas`;
        }

        function saveCustomPost() {
            const file = document.getElementById('custom-post-file').files[0];
            const url = document.getElementById('custom-post-url').value.trim();
            const caption = document.getElementById('custom-post-caption').value.trim();

            if (file) {
                const reader = new FileReader();
                reader.onload = function(event) {
                    customProfile.posts.push({ img: event.target.result, caption: caption });
                    saveCustomProfileData();
                    renderCustomProfile();
                    closeAddCustomPostModal();
                };
                reader.readAsDataURL(file);
            } else if (url) {
                customProfile.posts.push({ img: url, caption: caption });
                saveCustomProfileData();
                renderCustomProfile();
                closeAddCustomPostModal();
            } else {
                showToast('Añade una foto o URL.');
            }

        function setupTheme() {
            const toggleBtn = document.getElementById('theme-toggle');
            const icon = document.getElementById('theme-icon');

            if (window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches) {
                document.documentElement.classList.add('dark');
                icon.className = 'ph-bold ph-sun text-lg';
            }

            toggleBtn.addEventListener('click', () => {
                document.documentElement.classList.toggle('dark');
                if (document.documentElement.classList.contains('dark')) {
                    icon.className = 'ph-bold ph-sun text-lg';
                } else {
                    icon.className = 'ph-bold ph-moon text-lg';
                }
            });
        }
